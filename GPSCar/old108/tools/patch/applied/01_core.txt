/*************************************************************************************************
*	函 数 名:	Encoder_DiagDump
*
*	函数功能:	【诊断】把 4 路编码器相关寄存器一次性打印出来
*
*	说    明:	某一路"转轮子没示数"时, 这一条打印可以一次分清三类原因:
*				  1) 信号没进 MCU / 引脚被板载外设占用 -> IDR 对应位不跟轮子变
*				  2) 引脚复用没生效(被别处重配)        -> MODER/AFR0 不是 AF2
*				  3) 定时器没跑 / 没配成编码器模式      -> CR1.CEN=0 或 SMCR.SMS!=3
*				调用: 电机自检(KEY4)阶段3每秒自动打印一次, 也可自己挂到串口命令。
*
*	期 望 值(拿正常工作的 TIM4/TIM8 当参照对比):
*				GPIOA->MODER : PA0/PA1/PA6/PA7 的 2bit = 10 (AF)
*				GPIOA->AFR[0]: 第 0/1/6/7 个半字节 = 2 (AF2)
*				TIMx->CR1    : bit0(CEN)=1
*				TIMx->SMCR   : SMS[2:0]=011 (编码器 TI1+TI2 双边沿)
*				TIMx->CCMR1  : CC1S[1:0]=01, CC2S[9:8]=01
*				TIMx->CCER   : bit0(CC1E)=1, bit4(CC2E)=1
*
*************************************************************************************************/
void Encoder_DiagDump(void)
{
    printf("\r\n[ENC-DIAG] GPIOA MODER=%08lX AFR0=%08lX IDR=%08lX\r\n",
           (unsigned long)GPIOA->MODER, (unsigned long)GPIOA->AFR[0],
           (unsigned long)GPIOA->IDR);
    printf("[ENC-DIAG] PA0/PA1/PA6/PA7 = %u %u %u %u  (手转轮子时应看到跳变)\r\n",
           (unsigned)((GPIOA->IDR >> 0) & 1u), (unsigned)((GPIOA->IDR >> 1) & 1u),
           (unsigned)((GPIOA->IDR >> 6) & 1u), (unsigned)((GPIOA->IDR >> 7) & 1u));
    printf("[ENC-DIAG] TIM5 CNT=%5lu CR1=%08lX SMCR=%08lX CCMR1=%08lX CCER=%04lX\r\n",
           (unsigned long)TIM5->CNT, (unsigned long)TIM5->CR1, (unsigned long)TIM5->SMCR,
           (unsigned long)TIM5->CCMR1, (unsigned long)TIM5->CCER);
    printf("[ENC-DIAG] TIM3 CNT=%5lu CR1=%08lX SMCR=%08lX CCMR1=%08lX CCER=%04lX\r\n",
           (unsigned long)TIM3->CNT, (unsigned long)TIM3->CR1, (unsigned long)TIM3->SMCR,
           (unsigned long)TIM3->CCMR1, (unsigned long)TIM3->CCER);
    printf("[ENC-DIAG] TIM4 CNT=%5lu CR1=%08lX SMCR=%08lX CCMR1=%08lX CCER=%04lX\r\n",
           (unsigned long)TIM4->CNT, (unsigned long)TIM4->CR1, (unsigned long)TIM4->SMCR,
           (unsigned long)TIM4->CCMR1, (unsigned long)TIM4->CCER);
    printf("[ENC-DIAG] TIM8 CNT=%5lu CR1=%08lX SMCR=%08lX CCMR1=%08lX CCER=%04lX\r\n",
           (unsigned long)TIM8->CNT, (unsigned long)TIM8->CR1, (unsigned long)TIM8->SMCR,
           (unsigned long)TIM8->CCMR1, (unsigned long)TIM8->CCER);
    printf("[ENC-DIAG] RCC APB1ENR=%08lX(bit1=TIM3 bit2=TIM4 bit3=TIM5) APB2ENR=%08lX(bit0=TIM8)\r\n",
           (unsigned long)RCC->APB1ENR, (unsigned long)RCC->APB2ENR);
    printf("[ENC-DIAG] ACC: ENC1=%ld ENC2=%ld ENC3=%ld ENC4=%ld\r\n",
           (long)s_total[0], (long)s_total[1], (long)s_total[2], (long)s_total[3]);
}

/*************************************************************************************************
*	函 数 名:	USART3_IRQHandler
*
*	函数功能:	LD19 雷达的空闲中断: 只负责把 IDLE 标志清掉
*
*	说    明:	【这是一个必须存在的中断函数】
*				Lidar_Init() 里使能了 USART3 的 IDLE 中断和 NVIC, 但工程里原来
*				只有 asrpro.c 定义过 USART3_IRQHandler, 而 asrpro.c 并没有加进
*				Keil 工程 -> 向量表里 USART3 落在启动文件的 [WEAK] Default_Handler
*				(B . 死循环)。只要雷达开始吐数据, 总线上一定出现空闲 -> IDLE 置位
*				-> 进死循环, 而且 IDLE 没人清, 中断会一直挂着 -> 整车像卡死。
*				清 IDLE 的顺序必须是"先读 SR 再读 DR", 这里 USART_GetITStatus()
*				读 SR、USART_ReceiveData() 读 DR, 正好满足要求(与 gps.c 同款写法)。
*				真正的数据搬运仍然由 Lidar_Process() 从 DMA 缓冲里做。
*
*************************************************************************************************/
void USART3_IRQHandler(void)
{
    if (USART_GetITStatus(USART3, USART_IT_IDLE) != RESET)
    {
        USART_ReceiveData(USART3);		/* 读 SR + DR, 清除 IDLE 标志(不能省) */
    }
}

#FILE MY/stm32f4xx_it.c
#ENC gbk
#OP after 0 #include "sdio_sd.h"
#include <stdio.h>
#OP repl 7 void HardFault_Handler(void)
void HardFault_Handler(void)
{
    /* [2026-11] 原来只是 while(1), 现场什么都看不到。
     * 现在把 Cortex-M4 的故障寄存器打出来(用当前 MSP 的 ISR 栈, 4KB 够用),
     * 然后把 PC/LR 也留一份在静态变量里, 调试器停在这里就能看到。
     * 常见判读: CFSR 的 bit[7:0]=MMFSR(存储器), bit[15:8]=BFSR(总线),
     *           bit[31:16]=UFSR(用法故障); HFSR bit30=FORCED; MMFAR/BFAR=出错地址。
     * 注意: 打印完仍然死循环, 保留现场给调试器, 不做自动复位。 */
    volatile uint32_t cfsr  = SCB->CFSR;
    volatile uint32_t hfsr  = SCB->HFSR;
    volatile uint32_t mmfar = SCB->MMFAR;
    volatile uint32_t bfar  = SCB->BFAR;

    printf("\r\n[HARDFAULT] CFSR=%08lX HFSR=%08lX MMFAR=%08lX BFAR=%08lX\r\n",
           (unsigned long)cfsr, (unsigned long)hfsr,
           (unsigned long)mmfar, (unsigned long)bfar);
    printf("[HARDFAULT] (MMFSR=%02lX BFSR=%02lX UFSR=%04lX)\r\n",
           (unsigned long)(cfsr & 0xFFu), (unsigned long)((cfsr >> 8) & 0xFFu),
           (unsigned long)((cfsr >> 16) & 0xFFFFu));

    while (1)
    {
    }
}

/**
  * @brief  Optional independent watchdog (IWDG). Disabled by default.
  * @note   Enable with APP_USE_IWDG=1 in bsp.c, then call App_IwdgFeed()
  *         periodically from a task that is guaranteed to keep running
  *         (the LED heartbeat is a good place).
  */
void App_IwdgInit(void);
void App_IwdgFeed(void);
#FILE BSP/Src/bsp.c
#ENC gbk
#OP after 0 MapCollent_Init();
    App_IwdgInit();         /* optional IWDG, no-op while APP_USE_IWDG == 0   */
#OP eof 0 .
/*----------------------------------------------------------------------------
 *  Optional independent watchdog
 *----------------------------------------------------------------------------
 *  Why it is off by default:
 *    - a wrong timeout turns every debug session into a reset loop;
 *    - the current hang sources (missing USART3 IRQ, blocking ultrasonic)
 *      are already fixed, so a watchdog is a safety net, not a fix.
 *
 *  When enabled, the IWDG is clocked by LSI (30~60 kHz, so the real timeout
 *  is roughly half..double of the nominal value below) and it CANNOT be
 *  stopped once started - only a power cycle or a reset clears it.
 *
 *  Feed it from exactly ONE place (see LedTask in MY/main.c). If that task
 *  stops running (higher priority task spins forever), the MCU resets.
 *--------------------------------------------------------------------------*/
#define APP_USE_IWDG        0       /* 1 = enable, 0 = disabled (default) */
#define IWDG_PRESCALER      4       /* 0:/4 1:/8 2:/16 3:/32 4:/64 5:/128 6:/256 */
#define IWDG_RELOAD         1000    /* LSI/64 = 500 Hz -> 1000 = 2 s nominal */

void App_IwdgInit(void)
{
#if APP_USE_IWDG
    IWDG->KR  = 0x5555;                 /* unlock PR/RLR                    */
    IWDG->PR  = IWDG_PRESCALER;
    IWDG->RLR = IWDG_RELOAD;
    IWDG->KR  = 0xAAAA;                 /* load the counter                 */
    IWDG->KR  = 0xCCCC;                 /* start: cannot be stopped anymore */
#endif
}

void App_IwdgFeed(void)
{
#if APP_USE_IWDG
    IWDG->KR = 0xAAAA;
#endif
}
